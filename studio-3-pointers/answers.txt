10. char* reverseString( char* input ){
    int length = 0;
    while (input[length] != '\0'){
        length++;
    } 
    //printf("%i\n",length);
    char* output = (char*)malloc( length+1 );
    
    for (int i = 0; i < length; i++){
        output[i] = input[length-1-i];
    }
    output[length] = '\0';

    return output; 
}
